"""Company lookup pipeline — the core orchestration module.

Executes the full intelligence-gathering pipeline for a single Norwegian company:
validate → fetch → resolve identity → extract facts → audit → publish.
"""

from __future__ import annotations

import asyncio
import logging
import time
import uuid
from datetime import date, datetime, timezone
from decimal import Decimal
from typing import Any

import httpx
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.adapters.base import AdapterResult, AuthorityLevel, ExtractedFact, SourceAdapter
from app.adapters.brreg_enheter import BrregEnheterAdapter
from app.adapters.brreg_roller import BrregRollerAdapter
from app.adapters.brreg_regnskap import BrregRegnskapAdapter
from app.adapters.brreg_underenheter import BrregUnderenheterAdapter
from app.adapters.website import WebsiteAdapter
from app.core.adversarial_auditor import AdversarialAuditor
from app.core.budget_manager import BudgetManager
from app.core.cache_manager import CacheManager
from app.core.contradiction_engine import ContradictionEngine
from app.core.evidence_graph import EvidenceGraphEngine
from app.core.identity_firewall import IdentityFirewall
from app.core.org_number import normalize_org_number
from app.core.temporal_engine import calculate_freshness
from app.models.company import (
    Company,
    CompanyAddress,
    CompanyAlias,
    CompanyEvent,
    CompanyFact,
    CompanyIndustry,
    CompanyRole,
    CorporateRelationship,
)
from app.models.financial import FinancialStatement
from app.models.source import Document

logger = logging.getLogger(__name__)


class CompanyPipeline:
    """Orchestrates all agents for a single company lookup."""

    def __init__(
        self,
        db: AsyncSession,
        http_client: httpx.AsyncClient,
        cache_manager: CacheManager,
        budget_manager: BudgetManager,
    ):
        self.db = db
        self.http_client = http_client
        self.cache = cache_manager
        self.budget = budget_manager
        self.identity_firewall = IdentityFirewall()
        self.evidence_graph = EvidenceGraphEngine()
        self.contradiction_engine = ContradictionEngine()
        self.auditor = AdversarialAuditor()

    # ------------------------------------------------------------------
    # Public entry point
    # ------------------------------------------------------------------

    async def run(self, org_number: str, force_refresh: bool = False) -> dict[str, Any]:
        """Execute the full company intelligence pipeline.

        Returns a dictionary suitable for ProofPassport serialization.
        """
        start_time = time.monotonic()
        run_id = str(uuid.uuid4())

        # 1. Validate org number
        org_number = normalize_org_number(org_number)
        logger.info("[%s] Pipeline START for %s", run_id[:8], org_number)

        # 2. Check existing profile (skip if force_refresh)
        existing = await self._get_existing_company(org_number)
        if existing and not force_refresh and existing.profile_status == "verified":
            recently_verified = (
                existing.last_verified_at
                and (datetime.now(timezone.utc) - existing.last_verified_at).total_seconds()
                < 3600
            )
            if recently_verified:
                logger.info("[%s] Using cached verified profile", run_id[:8])
                return await self._build_passport_from_db(existing, run_id, start_time)

        # 3. STAGE 1 — BRREG Enheter (canonical identity)
        enheter_adapter = BrregEnheterAdapter(self.http_client, self.cache, self.budget)
        enheter_result = await enheter_adapter.fetch(org_number)

        if not enheter_result.success or enheter_result.status_code == 404:
            logger.warning("[%s] Company %s not found in BRREG", run_id[:8], org_number)
            return {
                "error": "not_found",
                "organization_number": org_number,
                "message": f"Organization number {org_number} not found in Enhetsregisteret.",
                "run_id": run_id,
            }

        raw_enheter = enheter_result.raw_data or {}
        enheter_facts = enheter_result.facts

        # 4. STAGE 2 — Identity Firewall
        identity_result = self.identity_firewall.resolve(
            requested_org_number=org_number,
            source_data={
                "org_number": raw_enheter.get("organisasjonsnummer"),
                "name": raw_enheter.get("navn"),
                "address": raw_enheter.get("forretningsadresse"),
            },
            source_adapter_id="brreg_enheter",
        )
        logger.info(
            "[%s] Identity: %s (%.2f)",
            run_id[:8],
            identity_result.decision,
            identity_result.confidence,
        )

        # 5. STAGE 3 — Parallel fetches
        roller_adapter = BrregRollerAdapter(self.http_client, self.cache, self.budget)
        regnskap_adapter = BrregRegnskapAdapter(self.http_client, self.cache, self.budget)
        underenheter_adapter = BrregUnderenheterAdapter(
            self.http_client, self.cache, self.budget
        )

        parallel_tasks = [
            roller_adapter.fetch(org_number),
            regnskap_adapter.fetch(org_number),
            underenheter_adapter.fetch(org_number),
        ]

        # Optionally fetch website
        homepage = raw_enheter.get("hjemmeside")
        if homepage:
            website_adapter = WebsiteAdapter(self.http_client, self.cache, self.budget)
            parallel_tasks.append(website_adapter.fetch(org_number, url=homepage))

        parallel_results = await asyncio.gather(*parallel_tasks, return_exceptions=True)

        roller_result = parallel_results[0] if not isinstance(parallel_results[0], Exception) else None
        regnskap_result = parallel_results[1] if not isinstance(parallel_results[1], Exception) else None
        underenheter_result = parallel_results[2] if not isinstance(parallel_results[2], Exception) else None
        website_result = (
            parallel_results[3]
            if len(parallel_results) > 3 and not isinstance(parallel_results[3], Exception)
            else None
        )

        # 6. STAGE 4 — Collect all facts
        all_facts: list[ExtractedFact] = list(enheter_facts)
        for result in [roller_result, regnskap_result, underenheter_result, website_result]:
            if result and isinstance(result, AdapterResult) and result.success:
                all_facts.extend(result.facts)

        # 7. STAGE 5 — Calculate freshness for each fact
        for fact in all_facts:
            fact_date = None
            if fact.source_date:
                try:
                    fact_date = date.fromisoformat(fact.source_date)
                except (ValueError, TypeError):
                    fact_date = None
            freshness = calculate_freshness(
                fact.fact_type, fact_date, fact.retrieved_at
            )

        # 8. STAGE 6 — Contradiction detection
        conflicts = self.contradiction_engine.detect(all_facts)

        # 9. STAGE 7 — Adversarial audit
        audit_reasons: list[str] = []
        published_facts: list[ExtractedFact] = []
        rejected_facts: list[ExtractedFact] = []

        for fact in all_facts:
            audit = self.auditor.audit_single(
                fact=fact,
                identity_decision=identity_result.decision,
                identity_confidence=identity_result.confidence,
            )
            if audit["publish"]:
                published_facts.append(fact)
            else:
                rejected_facts.append(fact)
                audit_reasons.extend(audit.get("reasons", []))

        # 10. STAGE 8 — Build evidence graph
        self.evidence_graph.clear()
        company_node = self.evidence_graph.add_company(org_number, raw_enheter.get("navn", ""))
        source_node = self.evidence_graph.add_source(
            "brreg_enheter", "Brønnøysund Enhetsregisteret", AuthorityLevel.GOVERNMENT
        )
        for i, fact in enumerate(published_facts):
            fact_node = self.evidence_graph.add_fact(
                f"fact_{i}", fact.fact_type, str(fact.value)
            )
            self.evidence_graph.link(company_node, fact_node, "HAS_FACT")
            self.evidence_graph.link(fact_node, source_node, "SUPPORTED_BY")

        graph_data = self.evidence_graph.to_serializable()

        # 11. STAGE 9 — Persist to database
        company = await self._persist_company(org_number, raw_enheter)
        await self._persist_addresses(company, raw_enheter)
        await self._persist_industries(company, raw_enheter)
        if roller_result and isinstance(roller_result, AdapterResult) and roller_result.success:
            await self._persist_roles(company, roller_result.raw_data or {})
        if regnskap_result and isinstance(regnskap_result, AdapterResult) and regnskap_result.success:
            await self._persist_financials(company, regnskap_result.raw_data or [])
        if underenheter_result and isinstance(underenheter_result, AdapterResult) and underenheter_result.success:
            await self._persist_subunits(company, underenheter_result.raw_data or {})

        # 12. STAGE 10 — Detect changes vs previous
        changes = await self._detect_changes(company, published_facts)

        # Mark verified
        company.profile_status = "verified"
        company.last_verified_at = datetime.now(timezone.utc)
        company.identity_confidence = identity_result.confidence

        try:
            await self.db.commit()
        except Exception:
            await self.db.rollback()
            logger.exception("[%s] Failed to commit", run_id[:8])

        elapsed = time.monotonic() - start_time
        budget_status = await self.budget.get_status()

        logger.info(
            "[%s] Pipeline COMPLETE for %s in %.2fs — %d facts published, %d rejected",
            run_id[:8],
            org_number,
            elapsed,
            len(published_facts),
            len(rejected_facts),
        )

        # 13. Assemble Proof Passport
        return self._assemble_passport(
            org_number=org_number,
            raw_enheter=raw_enheter,
            published_facts=published_facts,
            identity_result=identity_result,
            conflicts=conflicts,
            audit_reasons=audit_reasons,
            graph_data=graph_data,
            changes=changes,
            roller_data=roller_result.raw_data if roller_result and isinstance(roller_result, AdapterResult) else None,
            regnskap_data=regnskap_result.raw_data if regnskap_result and isinstance(regnskap_result, AdapterResult) else None,
            underenheter_data=underenheter_result.raw_data if underenheter_result and isinstance(underenheter_result, AdapterResult) else None,
            run_id=run_id,
            elapsed=elapsed,
            budget_status=budget_status,
        )

    # ------------------------------------------------------------------
    # Persistence helpers
    # ------------------------------------------------------------------

    async def _get_existing_company(self, org_number: str) -> Company | None:
        result = await self.db.execute(
            select(Company).where(Company.org_number == org_number)
        )
        return result.scalar_one_or_none()

    async def _persist_company(self, org_number: str, data: dict[str, Any]) -> Company:
        company = await self._get_existing_company(org_number)
        if not company:
            company = Company(org_number=org_number)
            self.db.add(company)

        company.name = data.get("navn", "")
        org_form = data.get("organisasjonsform") or {}
        company.org_form_code = org_form.get("kode")
        company.org_form_description = org_form.get("beskrivelse")

        reg_date = data.get("registreringsdatoEnhetsregisteret")
        if reg_date:
            try:
                company.registration_date = date.fromisoformat(reg_date)
            except (ValueError, TypeError):
                pass

        found_date = data.get("stiftelsesdato")
        if found_date:
            try:
                company.foundation_date = date.fromisoformat(found_date)
            except (ValueError, TypeError):
                pass

        del_date = data.get("slettedato")
        if del_date:
            try:
                company.deletion_date = date.fromisoformat(del_date)
            except (ValueError, TypeError):
                pass

        company.bankruptcy = bool(data.get("konkurs"))
        company.under_liquidation = bool(data.get("underAvvikling"))
        company.forced_liquidation = bool(
            data.get("underTvangsavviklingEllerTvangsopplosning")
        )
        company.registered_in_vat = data.get("registrertIMvaregisteret")
        company.registered_in_business_register = data.get("registrertIForetaksregisteret")
        company.homepage = data.get("hjemmeside")
        company.language_form = data.get("maalform")

        employees = data.get("antallAnsatte")
        if employees is not None:
            company.latest_annual_accounts_year = (
                int(data["sisteInnsendteAarsregnskap"])
                if data.get("sisteInnsendteAarsregnskap")
                else None
            )

        sect = data.get("institusjonellSektorkode") or {}
        company.institutional_sector_code = sect.get("kode")
        company.institutional_sector_desc = sect.get("beskrivelse")

        # Determine status
        if company.deletion_date:
            company.status = "dissolved"
        elif company.bankruptcy:
            company.status = "bankrupt"
        elif company.under_liquidation or company.forced_liquidation:
            company.status = "liquidating"
        else:
            company.status = "active"

        company.profile_status = "retrieved"
        company.updated_at = datetime.now(timezone.utc)

        await self.db.flush()
        return company

    async def _persist_addresses(self, company: Company, data: dict[str, Any]) -> None:
        for addr_type, key in [("business", "forretningsadresse"), ("postal", "postadresse")]:
            addr_data = data.get(key)
            if not addr_data:
                continue
            address = CompanyAddress(
                company_id=company.id,
                address_type=addr_type,
                street_lines=addr_data.get("adresse", []),
                postal_code=addr_data.get("postnummer"),
                city=addr_data.get("poststed"),
                municipality_code=addr_data.get("kommunenummer"),
                municipality_name=addr_data.get("kommune"),
                country_code=addr_data.get("landkode", "NO"),
                country_name=addr_data.get("land", "Norge"),
                is_current=True,
            )
            self.db.add(address)

    async def _persist_industries(self, company: Company, data: dict[str, Any]) -> None:
        for i in range(1, 4):
            nk = data.get(f"naeringskode{i}")
            if nk:
                ind = CompanyIndustry(
                    company_id=company.id,
                    priority=i,
                    nace_code=nk.get("kode", ""),
                    nace_description=nk.get("beskrivelse"),
                )
                self.db.add(ind)

    async def _persist_roles(self, company: Company, data: dict[str, Any]) -> None:
        for group in data.get("rollegrupper", []):
            group_type = group.get("type", {})
            for role in group.get("roller", []):
                role_type = role.get("type", {})
                person = role.get("person", {})
                entity = role.get("enhet", {})
                person_name = person.get("navn", {})

                cr = CompanyRole(
                    company_id=company.id,
                    role_group_code=group_type.get("kode"),
                    role_group_description=group_type.get("beskrivelse"),
                    role_type_code=role_type.get("kode", ""),
                    role_type_description=role_type.get("beskrivelse"),
                    person_first_name=person_name.get("fornavn"),
                    person_last_name=person_name.get("etternavn"),
                    person_middle_name=person_name.get("mellomnavn"),
                    is_resigned=bool(role.get("fratradt")),
                    is_deceased=bool(person.get("erDoed")),
                    entity_org_number=entity.get("organisasjonsnummer"),
                    entity_name=(entity.get("navn", [None])[0] if entity.get("navn") else None),
                    last_changed=(
                        date.fromisoformat(group.get("sistEndret"))
                        if group.get("sistEndret")
                        else None
                    ),
                )
                bd = person.get("fodselsdato")
                if bd:
                    try:
                        cr.person_birth_date = date.fromisoformat(bd)
                    except (ValueError, TypeError):
                        pass
                self.db.add(cr)

    async def _persist_financials(
        self, company: Company, data: list[dict[str, Any]] | dict[str, Any]
    ) -> None:
        records = data if isinstance(data, list) else [data]
        for rec in records:
            period = rec.get("regnskapsperiode", {})
            period_to = period.get("tilDato")
            if not period_to:
                continue
            try:
                fiscal_year = int(period_to[:4])
            except (ValueError, TypeError):
                continue

            # Navigate nested structure safely
            resultat = rec.get("resultatregnskap", {}) or {}
            dr = resultat.get("driftsresultat", {}) or {}
            di = dr.get("driftsinntekter", {}) or {}
            dk = dr.get("driftskostnader", {}) or {}
            fr = resultat.get("finansresultat", {}) or {}
            fi = fr.get("finansinntekter", {}) or {}
            fk = fr.get("finanskostnader", {}) or {}
            eiendeler = rec.get("eiendeler", {}) or {}
            om = eiendeler.get("omloepsmidler", {}) or {}
            ekg = rec.get("egenkapitalGjeld", {}) or {}
            ek = ekg.get("egenkapital", {}) or {}
            gjeld = ekg.get("gjeld", {}) or {}
            lg = gjeld.get("langsiktigGjeld", {}) or {}
            kg = gjeld.get("kortsiktigGjeld", {}) or {}

            def _dec(v: Any) -> Decimal | None:
                if v is None:
                    return None
                try:
                    return Decimal(str(v))
                except Exception:
                    return None

            fs = FinancialStatement(
                company_id=company.id,
                fiscal_year=fiscal_year,
                statement_type=rec.get("regnskapstype", "SELSKAP"),
                currency=rec.get("valuta", "NOK"),
                revenue=_dec(di.get("sumDriftsinntekter")),
                operating_profit=_dec(dr.get("driftsresultat")),
                net_income=_dec(resultat.get("aarsresultat")),
                total_assets=_dec(eiendeler.get("sumEiendeler")),
                equity=_dec(ek.get("sumEgenkapital")),
                total_debt=_dec(gjeld.get("sumGjeld")),
                current_assets=_dec(om.get("sumOmloepsmidler")),
                current_liabilities=_dec(kg.get("sumKortsiktigGjeld")),
                long_term_debt=_dec(lg.get("sumLangsiktigGjeld")),
                cash_and_equivalents=_dec(om.get("bankinnskuddKontanterOgLignende")),
                salary_costs=_dec(dk.get("lonnskostnad")),
                financial_income=_dec(fi.get("sumFinansinntekter")),
                financial_expenses=_dec(fk.get("sumFinanskostnader")),
                tax_expense=_dec(resultat.get("skattekostnadPaaOrdinaertResultat")),
                ebt=_dec(resultat.get("ordinaertResultatFoerSkattekostnad")),
                raw_data=rec,
                source_url=f"https://data.brreg.no/regnskapsregisteret/regnskap/{company.org_number}",
                retrieved_at=datetime.now(timezone.utc),
            )
            try:
                period_from = period.get("fraDato")
                if period_from:
                    fs.period_from = date.fromisoformat(period_from)
                if period_to:
                    fs.period_to = date.fromisoformat(period_to)
            except (ValueError, TypeError):
                pass

            self.db.add(fs)

    async def _persist_subunits(
        self, company: Company, data: dict[str, Any]
    ) -> None:
        embedded = data.get("_embedded", {})
        subunits = embedded.get("underenheter", [])
        for sub in subunits:
            sub_org = sub.get("organisasjonsnummer")
            if not sub_org:
                continue
            # Persist sub-unit as a company
            sub_company = await self._get_existing_company(sub_org)
            if not sub_company:
                sub_company = Company(
                    org_number=sub_org,
                    name=sub.get("navn", ""),
                    org_form_code=sub.get("organisasjonsform", {}).get("kode"),
                    profile_status="seeded",
                    status="active",
                )
                self.db.add(sub_company)
                await self.db.flush()

            rel = CorporateRelationship(
                parent_company_id=company.id,
                child_company_id=sub_company.id,
                relationship_type="BRANCH_OF",
            )
            self.db.add(rel)

    async def _detect_changes(
        self, company: Company, new_facts: list[ExtractedFact]
    ) -> list[dict[str, Any]]:
        """Compare new facts with stored facts to detect changes."""
        changes: list[dict[str, Any]] = []
        if not company.id:
            return changes

        result = await self.db.execute(
            select(CompanyFact).where(
                CompanyFact.company_id == company.id, CompanyFact.is_current == True  # noqa: E712
            )
        )
        existing = {f.predicate: f for f in result.scalars().all()}

        for fact in new_facts:
            old = existing.get(fact.predicate)
            if old and old.value_text != str(fact.value):
                change_type = "IDENTITY" if fact.fact_type in ("legal_name", "org_form") else "LEGAL"
                if fact.fact_type in ("registered_address", "postal_address"):
                    change_type = "LOCATION"
                elif fact.fact_type in ("board_member", "ceo"):
                    change_type = "PEOPLE"
                elif fact.fact_type == "employee_count":
                    change_type = "LEGAL"
                elif "financial" in fact.fact_type:
                    change_type = "FINANCIAL"

                changes.append({
                    "change_type": change_type,
                    "field": fact.predicate,
                    "old_value": old.value_text,
                    "new_value": str(fact.value),
                    "source": fact.source_adapter_id,
                    "confidence": fact.confidence,
                    "detected_at": datetime.now(timezone.utc).isoformat(),
                })

                # Create event
                event = CompanyEvent(
                    company_id=company.id,
                    event_type=change_type,
                    event_date=date.today(),
                    description=f"{fact.predicate}: {old.value_text} → {fact.value}",
                    old_value=old.value_text,
                    new_value=str(fact.value),
                )
                self.db.add(event)

        return changes

    async def _build_passport_from_db(
        self, company: Company, run_id: str, start_time: float
    ) -> dict[str, Any]:
        """Build passport from cached database records."""
        return self._assemble_passport(
            org_number=company.org_number,
            raw_enheter={},
            published_facts=[],
            identity_result=None,
            conflicts=[],
            audit_reasons=[],
            graph_data={"nodes": [], "edges": []},
            changes=[],
            roller_data=None,
            regnskap_data=None,
            underenheter_data=None,
            run_id=run_id,
            elapsed=time.monotonic() - start_time,
            budget_status=None,
        )

    # ------------------------------------------------------------------
    # Passport assembly
    # ------------------------------------------------------------------

    def _assemble_passport(
        self,
        org_number: str,
        raw_enheter: dict[str, Any],
        published_facts: list[ExtractedFact],
        identity_result: Any,
        conflicts: list[Any],
        audit_reasons: list[str],
        graph_data: dict[str, Any],
        changes: list[dict[str, Any]],
        roller_data: dict[str, Any] | None,
        regnskap_data: Any,
        underenheter_data: dict[str, Any] | None,
        run_id: str,
        elapsed: float,
        budget_status: Any,
    ) -> dict[str, Any]:
        """Assemble the final Proof Passport response."""
        now = datetime.now(timezone.utc).isoformat()

        # Build facts list for response
        response_facts = []
        for fact in published_facts:
            response_facts.append({
                "fact_type": fact.fact_type,
                "subject": fact.subject,
                "predicate": fact.predicate,
                "display_value": str(fact.value),
                "value_numeric": float(fact.value) if isinstance(fact.value, (int, float, Decimal)) else None,
                "source_name": fact.source_adapter_id,
                "source_url": fact.source_url,
                "source_date": fact.source_date,
                "retrieved_at": fact.retrieved_at.isoformat(),
                "confidence": fact.confidence,
                "identity_match_score": fact.identity_match_score,
                "fact_category": fact.fact_category,
                "why_this_company": (
                    f"Organization number {org_number} verified via direct API lookup "
                    f"at {fact.source_url}"
                ),
            })

        # Build roles from raw data
        roles = []
        if roller_data:
            for group in roller_data.get("rollegrupper", []):
                for role in group.get("roller", []):
                    person = role.get("person", {})
                    pname = person.get("navn", {})
                    entity = role.get("enhet", {})
                    roles.append({
                        "role_type_code": role.get("type", {}).get("kode", ""),
                        "role_type_description": role.get("type", {}).get("beskrivelse", ""),
                        "person_first_name": pname.get("fornavn"),
                        "person_last_name": pname.get("etternavn"),
                        "person_birth_date": person.get("fodselsdato"),
                        "is_resigned": bool(role.get("fratradt")),
                        "entity_name": (entity.get("navn", [None])[0] if entity.get("navn") else None),
                    })

        # Build financial summary
        financial_summary = None
        if regnskap_data:
            records = regnskap_data if isinstance(regnskap_data, list) else [regnskap_data]
            statements = []
            for rec in records:
                resultat = rec.get("resultatregnskap", {}) or {}
                dr = resultat.get("driftsresultat", {}) or {}
                di = dr.get("driftsinntekter", {}) or {}
                eiendeler = rec.get("eiendeler", {}) or {}
                ekg = rec.get("egenkapitalGjeld", {}) or {}
                ek = ekg.get("egenkapital", {}) or {}
                gjeld = ekg.get("gjeld", {}) or {}
                period = rec.get("regnskapsperiode", {})
                period_to = period.get("tilDato", "")
                try:
                    fy = int(period_to[:4])
                except (ValueError, TypeError, IndexError):
                    continue
                statements.append({
                    "fiscal_year": fy,
                    "statement_type": rec.get("regnskapstype", "SELSKAP"),
                    "currency": rec.get("valuta", "NOK"),
                    "revenue": di.get("sumDriftsinntekter"),
                    "operating_profit": dr.get("driftsresultat"),
                    "net_income": resultat.get("aarsresultat"),
                    "total_assets": eiendeler.get("sumEiendeler"),
                    "equity": ek.get("sumEgenkapital"),
                    "total_debt": gjeld.get("sumGjeld"),
                    "source_name": "Brønnøysund Regnskapsregisteret",
                    "source_url": f"https://data.brreg.no/regnskapsregisteret/regnskap/{org_number}",
                })
            if statements:
                financial_summary = {
                    "statements": statements,
                    "latest_year": max(s["fiscal_year"] for s in statements),
                    "has_group_accounts": any(s["statement_type"] == "KONSERN" for s in statements),
                }

        # Build addresses
        addresses = []
        for addr_type, key in [("business", "forretningsadresse"), ("postal", "postadresse")]:
            addr = raw_enheter.get(key)
            if addr:
                addresses.append({
                    "address_type": addr_type,
                    "street_lines": addr.get("adresse", []),
                    "postal_code": addr.get("postnummer"),
                    "city": addr.get("poststed"),
                    "municipality_name": addr.get("kommune"),
                })

        # Build industries
        industries = []
        for i in range(1, 4):
            nk = raw_enheter.get(f"naeringskode{i}")
            if nk:
                industries.append({
                    "priority": i,
                    "nace_code": nk.get("kode", ""),
                    "nace_description": nk.get("beskrivelse"),
                })

        # Build subsidiaries
        subsidiaries = []
        if underenheter_data:
            embedded = underenheter_data.get("_embedded", {})
            for sub in embedded.get("underenheter", []):
                subsidiaries.append({
                    "org_number": sub.get("organisasjonsnummer", ""),
                    "name": sub.get("navn", ""),
                    "org_form_code": sub.get("organisasjonsform", {}).get("kode"),
                })

        # Conflict serialization
        conflict_list = []
        for c in conflicts:
            conflict_list.append({
                "fact_type": getattr(c, "fact_type", str(c)),
                "values": getattr(c, "values", []),
                "resolution": getattr(c, "resolution", "UNRESOLVED"),
                "resolution_reasoning": getattr(c, "reasoning", None),
            })

        # Identity match serialization
        identity_dict = None
        if identity_result:
            identity_dict = {
                "organization_number": org_number,
                "canonical_name": raw_enheter.get("navn", ""),
                "match_score": identity_result.confidence,
                "decision": identity_result.decision,
                "reasoning": f"Organization number {org_number} directly matched in BRREG API response.",
                "matched_signals": [
                    {
                        "signal_type": s.field,
                        "matches": s.status == "MATCH",
                        "confidence": s.match_score,
                    }
                    for s in identity_result.signals
                ],
                "conflicting_signals": [
                    {
                        "signal_type": s.field,
                        "matches": False,
                        "confidence": s.match_score,
                    }
                    for s in identity_result.signals
                    if s.status == "MISMATCH"
                ],
            }

        org_form = raw_enheter.get("organisasjonsform", {})
        budget_dict = None
        if budget_status:
            budget_dict = {
                "requests_used": budget_status.requests_used,
                "requests_remaining": budget_status.remaining_requests,
                "requests_saved_by_cache": budget_status.requests_saved_by_cache,
                "cache_hit_rate": (
                    budget_status.requests_saved_by_cache
                    / max(budget_status.requests_used + budget_status.requests_saved_by_cache, 1)
                ),
                "external_cost_estimate": float(budget_status.cost_estimate),
                "cost_remaining": float(budget_status.remaining_budget),
            }

        # Source ledger
        sources_consulted = [
            {
                "source_name": "Brønnøysund Enhetsregisteret",
                "source_url": f"https://data.brreg.no/enhetsregisteret/api/enheter/{org_number}",
                "authority_level": 1,
                "retrieved_at": now,
                "facts_extracted": len([f for f in published_facts if f.source_adapter_id == "brreg_enheter"]),
            },
            {
                "source_name": "Brønnøysund Roller",
                "source_url": f"https://data.brreg.no/enhetsregisteret/api/enheter/{org_number}/roller",
                "authority_level": 1,
                "retrieved_at": now,
                "facts_extracted": len([f for f in published_facts if f.source_adapter_id == "brreg_roller"]),
            },
            {
                "source_name": "Brønnøysund Regnskapsregisteret",
                "source_url": f"https://data.brreg.no/regnskapsregisteret/regnskap/{org_number}",
                "authority_level": 1,
                "retrieved_at": now,
                "facts_extracted": len([f for f in published_facts if f.source_adapter_id == "brreg_regnskap"]),
            },
        ]

        return {
            "company_name": raw_enheter.get("navn", ""),
            "organization_number": org_number,
            "organization_type": org_form.get("beskrivelse"),
            "organization_type_code": org_form.get("kode"),
            "status": (
                "bankrupt" if raw_enheter.get("konkurs")
                else "dissolved" if raw_enheter.get("slettedato")
                else "liquidating" if raw_enheter.get("underAvvikling")
                else "active"
            ),
            "identity_confidence": identity_result.confidence if identity_result else 1.0,
            "last_verified": now,
            "proof_score": len(published_facts) / max(len(published_facts) + len(rejected_facts), 1),
            "facts": response_facts,
            "financial_summary": financial_summary,
            "roles": roles,
            "addresses": addresses,
            "industries": industries,
            "parent": None,
            "subsidiaries": subsidiaries,
            "identity_match": identity_dict,
            "conflicts": conflict_list,
            "timeline": [],
            "evidence_graph": graph_data,
            "sources_consulted": sources_consulted,
            "changes_detected": changes,
            "budget": budget_dict,
            "run_id": run_id,
            "retrieval_time_sec": round(elapsed, 3),
            "requests_made": budget_status.requests_used if budget_status else 0,
            "cache_hits": budget_status.requests_saved_by_cache if budget_status else 0,
        }
